# Pasar a formato largo
datos_long <- datos %>%
  pivot_longer(cols = c(Hogares, Comercial, Gobierno, Otros),
               names_to = "Dependencia",
               values_to = "Lineas")

# Calcular extremos y variación %
extremos <- datos_long %>%
  group_by(Dependencia) %>%
  summarise(
    x_ini = min(Año),
    x_fin = max(Año),
    y_ini = first(Lineas),
    y_fin = last(Lineas),
    Variacion_fmt = sprintf("%.2f%%", (y_fin - y_ini) / y_ini * 100),
    .groups = "drop"
  )

# Gráfico facetado con línea y etiqueta
ggplot(datos_long, aes(x = Año, y = Lineas, color = Dependencia)) +
  geom_line(size = 1) +
  geom_segment(data = extremos,
               aes(x = x_ini, xend = x_fin,
                   y = y_ini, yend = y_fin),
               linetype = "dashed", size = 1.2,
               color = "black", inherit.aes = FALSE) +
  geom_text(data = extremos,
            aes(x = x_fin - 1, y = y_fin,
                label = Variacion_fmt),
            color = "black", fontface = "bold",
            size = 3,
            hjust = 0.7, vjust = -5,
            inherit.aes = FALSE) +
  facet_wrap(~Dependencia, scales = "free_y", ncol = 2) +
  scale_y_continuous(labels = comma_format(big.mark = ".", decimal.mark = ",")) +
  scale_color_brewer(palette = "Set1") +
  labs(title = "Evolución de líneas fijas por dependencia",
       x = "Año", y = "Cantidad de líneas") +
  theme_minimal(base_size = 13) +
  theme(legend.position = "none",
        strip.text = element_text(face = "bold"))
